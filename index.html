function doGet(e) { var props = PropertiesService.getScriptProperties(); var action = e.parameter.action; if (action === 'save_chunk') { props.setProperty('chunk_' + e.parameter.index, e.parameter.data); return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON); } if (action === 'save_commit') { var total = parseInt(e.parameter.total, 10); var full = ''; for (var i = 0; i < total; i++) { full += props.getProperty('chunk_' + i) || ''; props.deleteProperty('chunk_' + i); } props.setProperty('calc_data', full); return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON); } var data = props.getProperty('calc_data') || '{}'; return ContentService.createTextOutput(data).setMimeType(ContentService.MimeType.JSON); }
